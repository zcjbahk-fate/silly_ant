  <script>
  document.addEventListener("DOMContentLoaded", function(){
    // ===== 原有选项逻辑（完全不变） =====
    function init(){
      const refs = { 
        card: document.getElementById("post-it-card"), 
        collapsibleContent: document.getElementById("collapsible-content"), 
        contentWrapper: document.querySelector(".content-wrapper"),
        optionsContainer: document.getElementById("options-container"), 
        rawData: document.getElementById("raw-options-data") 
      };
      if(refs.card && refs.rawData && refs.optionsContainer){
        renderOptions(refs.rawData, refs.optionsContainer);
        bindCollapsible(refs);
      }
    }

    function renderOptions(rawNode, root){
      const lines = rawNode.textContent.trim().split(/\r?\n/).filter(s=>s.trim()!=="");
      lines.forEach((line)=>{
        const text = line.trim().replace(/^[A-Z]\.\s*/i,"");
        if(!text) return;
        const item = document.createElement("div");
        item.className = "clickable-text";
        item.setAttribute("data-action", text);
        item.innerHTML = `<li>${text}</li>`;
        item.addEventListener("click", function(e){
          e.stopPropagation();
          const rect = this.getBoundingClientRect();
          const x = e.clientX - rect.left;
          const y = e.clientY - rect.top;
          this.style.setProperty('--rx', x + 'px');
          this.style.setProperty('--ry', y + 'px');
          this.classList.add('is-selected');
          setTimeout(()=>{ this.classList.remove('is-selected'); }, 700);
          const payload = this.getAttribute("data-action");
          try {
            const t = window.parent.document.querySelector("#send_textarea");
            const n = window.parent.triggerSlash;
            if(t){
              const v = t.value + payload;
              if(n){ n(`/setinput ${v}`); }
              else{ t.value = v; t.dispatchEvent(new Event("input",{bubbles:true})); }
            } else if(n){ n(`/setinput ${payload}`); }
          } catch(err){ console.error("设置输入文本时出错:", err); }
        });
        root.appendChild(item);
      });
    }

    function bindCollapsible(refs){
      const { collapsibleContent, contentWrapper, optionsContainer } = refs;
      contentWrapper.addEventListener("click", e => {
        const exceptedElements = [...optionsContainer.querySelectorAll(".clickable-text")];
        const isClickOnException = exceptedElements.some(el => el && el.contains(e.target));
        if(!isClickOnException){ collapsibleContent.classList.toggle("open"); }
      });
    }
    init();

    // ===== 播放器逻辑（唱片显示已修正，封面更新） =====
    (function setupMusicPlayer() {
      // 从隐藏的XML中提取歌曲名
      function getSongNameFromXML() {
        const xmlDiv = document.getElementById('music-data');
        if (!xmlDiv) return '$2';
        const content = xmlDiv.textContent;
        const match = content.match(/<当前播放>(.*?)<\/当前播放>/);
        return match ? match[1].trim() : '$2';
      }

      const SONG_NAME = getSongNameFromXML(); // 例如 "晴天"

      // 获取DOM元素
      const vinyl = document.getElementById('vinyl');
      const playBtn = document.getElementById('playPauseBtn');
      const playIcon = document.getElementById('playIcon');
      const pauseIcon = document.getElementById('pauseIcon');
      const progressSlider = document.getElementById('progressSlider');
      const volumeSlider = document.getElementById('volumeSlider');
      const currentTimeSpan = document.getElementById('currentTime');
      const totalTimeSpan = document.getElementById('totalTime');
      const songTitleEl = document.getElementById('songNamePlaceholder');
      const artistEl = document.getElementById('artistPlaceholder');
      const coverImage = document.getElementById('coverImage');

      // 音频元素
      const audio = new Audio();
      audio.volume = 0.8;

      // 状态变量
      let isPlaying = false;
      let songData = null;
      let lastSearchedSong = '';

      // 格式化时间
      function formatTime(seconds) {
        if (isNaN(seconds) || seconds < 0) return '0:00';
        const mins = Math.floor(seconds / 60);
        const secs = Math.floor(seconds % 60);
        return `${mins}:${secs < 10 ? '0' : ''}${secs}`;
      }

      // 更新播放状态UI（使用 playing 类）
      function updatePlayState(playing) {
        if (playing) {
          playIcon.style.display = 'none';
          pauseIcon.style.display = 'block';
          vinyl.classList.add('playing');
        } else {
          playIcon.style.display = 'block';
          pauseIcon.style.display = 'none';
          vinyl.classList.remove('playing');
        }
      }

      // 切换播放/暂停
      function togglePlay() {
        if (!songData || !songData.Url) return;
        if (audio.paused) {
          audio.play().then(() => {
            isPlaying = true;
            updatePlayState(true);
          }).catch(e => console.warn('播放失败:', e));
        } else {
          audio.pause();
          isPlaying = false;
          updatePlayState(false);
        }
      }

      // 根据歌曲名搜索
      async function searchAndLoadSong(songName) {
        if (!songName || songName === lastSearchedSong || songName.startsWith('$')) return;

        lastSearchedSong = songName;
        songTitleEl.textContent = songName;
        artistEl.textContent = '搜索中...';

        try {
          if (typeof Music !== 'undefined' && Music.SearchMusic) {
            const result = await Music.SearchMusic(songName);
            if (result && result.Url) {
              songData = result;
              audio.src = result.Url;
              audio.load();
              songTitleEl.textContent = result.Name || songName;
              artistEl.textContent = result.Singer || '未知歌手';
              // 更新封面
              if (result.Cover) {
                coverImage.src = result.Cover;
              }
            } else {
              songData = null;
              artistEl.textContent = '暂无歌曲';
            }
          } else {
            songData = null;
            artistEl.textContent = '暂无歌曲';
          }
        } catch (error) {
          console.error('搜索音乐失败:', error);
          songData = null;
          artistEl.textContent = '暂无歌曲';
        }
      }

      // 初始加载
      if (SONG_NAME && !SONG_NAME.startsWith('$')) {
        searchAndLoadSong(SONG_NAME);
      } else {
        songTitleEl.textContent = SONG_NAME;
        artistEl.textContent = '暂无歌曲';
      }

      // 音频事件监听
      audio.addEventListener('timeupdate', () => {
        if (!isNaN(audio.duration)) {
          progressSlider.value = audio.currentTime;
          progressSlider.max = audio.duration;
          currentTimeSpan.textContent = formatTime(audio.currentTime);
        }
      });

      audio.addEventListener('loadedmetadata', () => {
        totalTimeSpan.textContent = formatTime(audio.duration);
        progressSlider.max = audio.duration;
      });

      audio.addEventListener('ended', () => {
        isPlaying = false;
        updatePlayState(false);
        audio.currentTime = 0;
      });

      // 进度条拖动
      progressSlider.addEventListener('input', (e) => {
        if (!songData || !songData.Url) return;
        const seekTime = parseFloat(e.target.value);
        audio.currentTime = seekTime;
        currentTimeSpan.textContent = formatTime(seekTime);
      });

      // 音量调节
      volumeSlider.addEventListener('input', (e) => {
        const vol = parseFloat(e.target.value);
        audio.volume = vol;
      });

      // 播放按钮点击
      playBtn.addEventListener('click', (e) => {
        e.stopPropagation();
        togglePlay();
      });

      // 阻止滑块点击事件冒泡
      progressSlider.addEventListener('click', e => e.stopPropagation());
      volumeSlider.addEventListener('click', e => e.stopPropagation());
    })();
  });
  </script>
</body>
