                </div>);
        };

        /* --- MAIN COMPONENT --- */
        const SwissDynamicIsland = () => {
            const islandRef = useRef(null);
            const [activeId, setActiveId] = useState(null);
            const [quoteIndex, setQuoteIndex] = useState(0);

            // --- 统一状态管理 ---
            const [isExpanded, setIsExpanded] = useState(() => { try { return localStorage.getItem('dsi_isExpanded') === 'true'; } catch { return false; } });
            const [fontScale, setFontScale] = useState(() => { try { return parseFloat(localStorage.getItem('dsi_font_scale')) || 1.0; } catch { return 1.0; } });
            const [islandWidth, setIslandWidth] = useState(() => { try { return parseFloat(localStorage.getItem('dsi_width')) || 100; } catch { return 100; } });
            const [islandHeight, setIslandHeight] = useState(() => { try { return parseInt(localStorage.getItem('dsi_height'), 10) || 680; } catch { return 680; } });
            const [volume, setVolume] = useState(() => { try { const v = localStorage.getItem('dsi_volume'); return v !== null ? parseFloat(v) : 0.8; } catch { return 0.8; } });
            
            // 用于记录静音前的音量，以便恢复
            const lastVolumeRef = useRef(0.8);

            const scaleToChinese = { "0.7": "壹", "0.8": "贰", "0.9": "叁", "1.0": "肆", "1.1": "伍", "1.2": "陆", "1.3": "柒", "1.4": "捌", "1.5": "玖", "1.6": "拾" };

            useEffect(() => {
                const node = islandRef.current;
                if (node) {
                    node.style.setProperty('--font-scale', fontScale);
                    node.style.setProperty('--island-width', `${islandWidth}%`);
                    node.style.setProperty('--island-height', `${islandHeight}px`);
                }
                try { 
                    localStorage.setItem('dsi_isExpanded', isExpanded);
                    localStorage.setItem('dsi_font_scale', fontScale.toFixed(2));
                    localStorage.setItem('dsi_width', islandWidth.toString());
                    localStorage.setItem('dsi_height', islandHeight.toString());
                    localStorage.setItem('dsi_volume', volume.toString());
                } catch (e) { console.warn('Failed to save settings:', e); }
            }, [isExpanded, fontScale, islandWidth, islandHeight, volume]);

            const handleIncreaseFont = (e) => { e.stopPropagation(); setFontScale(prev => Math.min(1.6, parseFloat((prev + 0.1).toFixed(1)))); };
            const handleDecreaseFont = (e) => { e.stopPropagation(); setFontScale(prev => Math.max(0.7, parseFloat((prev - 0.1).toFixed(1)))); };
            const handleWidthChange = (e) => { e.stopPropagation(); setIslandWidth(parseFloat(e.target.value)); };
            const handleHeightChange = (e) => { e.stopPropagation(); setIslandHeight(parseInt(e.target.value, 10)); };

            // --- 其他状态管理 ---
            const [charName, setCharName] = useState(() => { try { return localStorage.getItem('dsi_name') || DEFAULT_NAME; } catch (e) { return DEFAULT_NAME; } });
            const [bgUrl, setBgUrl] = useState(() => { try { return localStorage.getItem('dsi_bg') || DEFAULT_BG; } catch (e) { return DEFAULT_BG; } });
            const [bgSaveStatus, setBgSaveStatus] = useState('');
            useEffect(() => { try { localStorage.setItem('dsi_name', charName); } catch (e) { console.warn('Failed to save name:', e); } }, [charName]);
            useEffect(() => { if (bgUrl === DEFAULT_BG || bgUrl.startsWith('http')) { try { localStorage.setItem('dsi_bg', bgUrl); setBgSaveStatus('saved'); } catch (e) { console.warn('Failed to save bg:', e); setBgSaveStatus('session-only'); } } }, [bgUrl]);

            // --- 音乐 & 交互逻辑 ---
            const [bubbles, setBubbles] = useState([]);
            const audioRef = useRef(null);
            const [songData, setSongData] = useState(null);
            const [isLoadingMusic, setIsLoadingMusic] = useState(false);
            const [isPlaying, setIsPlaying] = useState(false);
            const [progress, setProgress] = useState(0);
            const [duration, setDuration] = useState(0);
            const [lastSearchedSong, setLastSearchedSong] = useState('');
            const searchAndLoadSong = async (songName) => { if (!songName || songName === lastSearchedSong || songName.startsWith('$')) return; setIsLoadingMusic(true); setLastSearchedSong(songName); try { if (typeof Music !== 'undefined' && Music.SearchMusic) { const result = await Music.SearchMusic(songName); if (result && result.Url) { setSongData(result); if (audioRef.current) { audioRef.current.src = result.Url; audioRef.current.load(); } } else { setSongData(null); } } else { console.warn('Music API not loaded'); setSongData(null); } } catch (error) { console.error('Failed to search music:', error); setSongData(null); } finally { setIsLoadingMusic(false); } };
            useEffect(() => { const songName = SONG_NAME_FROM_STATUS; if (songName && !songName.startsWith('$')) { searchAndLoadSong(songName); } }, []);
            useEffect(() => { const audio = audioRef.current; if (!audio) return; audio.volume = volume; const updateProgress = () => setProgress(audio.currentTime); const setAudioDuration = () => setDuration(audio.duration); const handleEnd = () => setIsPlaying(false); const handleCanPlay = () => setIsLoadingMusic(false); audio.addEventListener('timeupdate', updateProgress); audio.addEventListener('loadedmetadata', setAudioDuration); audio.addEventListener('ended', handleEnd); audio.addEventListener('canplay', handleCanPlay); return () => { audio.removeEventListener('timeupdate', updateProgress); audio.removeEventListener('loadedmetadata', setAudioDuration); audio.removeEventListener('ended', handleEnd); audio.removeEventListener('canplay', handleCanPlay); }; }, [volume]);
            const togglePlay = (e) => { if(e) e.stopPropagation(); const audio = audioRef.current; if (!audio || !songData?.Url) return; if (isPlaying) audio.pause(); else audio.play(); setIsPlaying(!isPlaying); };
            const handleSeek = (e) => { const audio = audioRef.current; if (!audio) return; const val = Number(e.target.value); audio.currentTime = val; setProgress(val); };
            
            // 音量控制逻辑
            const handleVolume = (e) => { const newVol = parseFloat(e.target.value); setVolume(newVol); };
            
            // 一键静音逻辑
            const handleToggleMute = () => {
                if (volume > 0) {
                    lastVolumeRef.current = volume; // 记住当前音量
                    setVolume(0);
                } else {
                    // 恢复之前的音量，如果之前也没有（比如初始就是0），则给个默认值
                    setVolume(lastVolumeRef.current || 0.5);
                }
            };

            const handleMouseMove = (e) => { if (isExpanded) return; const rect = e.currentTarget.getBoundingClientRect(); const x = e.clientX - rect.left; const y = e.clientY - rect.top; e.currentTarget.style.setProperty('--mouse-x', `${x}px`); e.currentTarget.style.setProperty('--mouse-y', `${y}px`); };
            const handleTriggerClick = (e) => { if (!isExpanded) return; const triggerZone = e.currentTarget; const rect = triggerZone.getBoundingClientRect(); const x = e.clientX - rect.left + triggerZone.offsetLeft; const y = e.clientY - rect.top + triggerZone.offsetTop; const text = CHAR_DIALOGUES.length > 0 ? CHAR_DIALOGUES[quoteIndex % CHAR_DIALOGUES.length] : "点击空白区域触发对话"; if (CHAR_DIALOGUES.length > 0) { setQuoteIndex((prev) => (prev + 1) % CHAR_DIALOGUES.length); } const bubbleId = Date.now(); const newBubble = { id: bubbleId, x: x, y: y, text: text, isFading: false }; setBubbles(prev => { const newQueue = [...prev, newBubble]; if (newQueue.length > 2) { return newQueue.slice(newQueue.length - 2); } return newQueue; }); setTimeout(() => { setBubbles(prev => prev.map(b => b.id === bubbleId ? { ...b, isFading: true } : b)); }, 5000); setTimeout(() => { setBubbles(prev => prev.filter(b => b.id !== bubbleId)); }, 6500); };
            const handleImageUpload = (e) => { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onloadend = () => { const base64Result = reader.result; setBgUrl(base64Result); try { localStorage.setItem('dsi_bg', base64Result); setBgSaveStatus('saved'); } catch (error) { console.warn('localStorage quota exceeded.', error); setBgSaveStatus('session-only'); alert('图片较大，无法永久保存。背景将在本次会话中生效，刷新页面后会恢复默认。'); } }; reader.onerror = () => { console.error('Failed to read file.'); alert('读取文件失败，请重试。'); }; reader.readAsDataURL(file); e.target.value = null; };

            // --- 内容定义 ---
            const searchRecords = parseSearchHistory(`$11`);
            const details = [
                { id: 'loc', icon: <i className="fa-solid fa-map-location-dot"></i>, title: 'LOCATION / TIME', content: ( <div className="info-stack"><div className="detail-row"><span className="detail-label">Coordinates</span><span className="detail-value">$2</span></div><div className="detail-row" style={{ borderBottom: 'none' }}><span className="detail-label">Local Time</span><span className="detail-value text-bold">$3</span></div><div className="text-desc">$4</div></div> ) },
                { id: 'outfit', icon: <i className="fa-solid fa-user-astronaut"></i>, title: 'OUTFIT / POSE', content: ( <div className="tag-grid"><div className="tag-box"><span className="detail-label mb-1">Upper</span>$5</div><div className="tag-box"><span className="detail-label mb-1">Lower</span>$6</div><div className="tag-box span-2"><span className="detail-label mb-1">Pose</span>$7</div></div> ) },
                { id: 'mood', icon: <i className="fa-solid fa-heart"></i>, title: 'CURRENT MOOD', content: ( <div className="mood-box"><div className="mood-circle">$8</div><div className="mood-info"><span className="text-bold" style={{ fontSize: '0.875rem' }}>$9</span><div className="text-desc" style={{ marginTop: '0.25rem', opacity: 0.7 }}>$10</div></div></div> ) },
                { id: 'search', icon: <i className="fa-solid fa-magnifying-glass"></i>, title: 'PRIVATE SEARCH', content: ( <div className="search-list"><div className="detail-row search-header"><span>QUERY</span><span>TIMESTAMP</span></div>{searchRecords.length > 0 ? ( searchRecords.map((record, idx) => ( <div key={idx} className="search-item"><span className="text-ellipsis">{record.content}</span><span className="text-faint">{record.time}</span></div> )) ) : ( <div className="text-desc" style={{textAlign: 'center', opacity: 0.5}}>暂无搜索记录</div> )}</div> ) },
                { id: 'music', icon: <i className="fa-solid fa-compact-disc"></i>, title: 'AUDIO LOG', content: <MusicPlayerUI songData={songData} isLoading={isLoadingMusic} isPlaying={isPlaying} progress={progress} duration={duration} volume={volume} onTogglePlay={togglePlay} onSeek={handleSeek} onVolumeChange={handleVolume} onToggleMute={handleToggleMute} /> },
                { id: 'settings', icon: <i className="fa-solid fa-gear"></i>, title: 'SYSTEM SETTINGS', content: ( <div className="info-stack settings-stack"><div className="detail-row" style={{ borderBottom: 'none', paddingBottom: 0 }}><span className="detail-label">Display Name</span></div><input type="text" className="setting-input" value={charName} onChange={(e) => setCharName(e.target.value)} placeholder="Enter name..." onClick={e => e.stopPropagation()} /><div className="detail-row" style={{ borderBottom: 'none', paddingBottom: 0, marginTop: '8px' }}><span className="detail-label">Background Image</span></div><label className="file-upload-label" onClick={e => e.stopPropagation()}><i className="fa-solid fa-cloud-arrow-up"></i><span>Upload</span><input type="file" accept="image/png, image/jpeg, image/gif" hidden onChange={handleImageUpload} /></label><div className="text-desc" style={{textAlign:'center', fontSize: '0.65rem'}}>{bgSaveStatus === 'session-only' ? 'Image too large for permanent save.' : 'Image saved in browser.'}</div><div className="slider-control-row"><div className="slider-control"><label className="slider-label">Width</label><input type="range" min="33.3" max="100" value={islandWidth} onChange={handleWidthChange} className="settings-slider" /></div><div className="slider-control"><label className="slider-label">Height</label><input type="range" min="453" max="900" value={islandHeight} onChange={handleHeightChange} className="settings-slider" /></div></div></div> ) }
            ];
            const activeItem = details.find(d => d.id === activeId);

            return (
                <div ref={islandRef} onClick={() => !isExpanded && setIsExpanded(true)} onMouseMove={handleMouseMove} className={`swiss-island ${isExpanded ? 'expanded' : 'collapsed'}`}>
                    <audio ref={audioRef} style={{display:'none'}} />
                    <div className="island-bg-image" style={{ backgroundImage: `url(${bgUrl})`, opacity: isExpanded ? 1 : 0 }}></div>
                    <div className="island-overlay" style={{ opacity: isExpanded ? 1 : 0 }}></div>
                    {isExpanded && <div className="scan-line"></div>}
                    {isExpanded && bubbles.map(bubble => ( <div key={bubble.id} className={`speech-bubble ${bubble.isFading ? 'fading' : ''}`} style={{ left: bubble.x, top: bubble.y - 20 }}>{bubble.text}</div> ))}
                    <div className="capsule-content"><span className="capsule-title">{charName}</span><span className="capsule-date">$1</span></div>
                    <div className="card-content">
                        <div className="bubble-trigger-zone" onClick={handleTriggerClick}></div>
                        <div className="card-header-row">
                            <div><h1 className="main-title">{charName}</h1><div className="status-badge"><div className="status-dot"></div><span className="status-text">ACTIVE</span></div></div>
                            <button onClick={(e) => { e.stopPropagation(); setIsExpanded(false); setActiveId(null); setBubbles([]); }} className="close-btn"><svg width="14" height="14" viewBox="0 0 14 14" fill="none" stroke="currentColor" strokeWidth="2"><path d="M1 1L13 13M1 13L13 1" /></svg></button>
                        </div>
                        {isExpanded && (<div className="font-size-controls"><button className="font-size-btn" onClick={handleIncreaseFont}><b>+</b></button><span className="font-size-display">{scaleToChinese[fontScale.toFixed(1)] || '肆'}</span><button className="font-size-btn" onClick={handleDecreaseFont}><b>-</b></button></div>)}
                        <div className="content-display-area" onClick={(e) => e.stopPropagation()}>{activeItem && (<div className="detail-glass-box" key={activeItem.id}><div className="detail-header-title">{activeItem.title}</div>{activeItem.content}</div>)}</div>
                        <div className="nav-footer" onClick={(e) => e.stopPropagation()}>{details.map((item) => (<button key={item.id} className={`nav-icon-btn ${activeId === item.id ? 'active' : ''}`} onClick={(e) => { e.stopPropagation(); setActiveId(activeId === item.id ? null : item.id); }}><span>{item.icon}</span><div className="nav-dot"></div></button>))}</div>
                    </div>
                </div>);
        };

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<SwissDynamicIsland />);
    </script>
</body>
