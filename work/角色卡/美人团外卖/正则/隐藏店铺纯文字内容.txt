    document.addEventListener('DOMContentLoaded', function() {
      const foldElements = document.querySelectorAll('.tavern-fold');

      foldElements.forEach(element => {
        const summary = element.querySelector('.tavern-fold-summary');

        const toggleFold = () => {
          element.classList.toggle('is-open');
          const isExpanded = element.classList.contains('is-open');
          summary.setAttribute('aria-expanded', isExpanded);
        };

        summary.addEventListener('click', toggleFold);

        // 增加键盘可访问性 (Enter 和 Space 键)
        summary.addEventListener('keydown', (e) => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            toggleFold();
          }
        });
      });
    });
