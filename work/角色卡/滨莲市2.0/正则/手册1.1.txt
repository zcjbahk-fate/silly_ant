  <script type="module">
    // 关键修改：从父页面(酒馆主页面)借用所有需要的全局变量
    const parentWindow = window.parent;
    const parent$ = parentWindow.$;

    // 通知父页面的 iframe 调整高度的函数
    function updateIframeHeight() {
      if (parentWindow && parentWindow !== window) {
        const bodyHeight = document.body.scrollHeight;
        parentWindow.postMessage({ type: 'resizeIframe', height: bodyHeight }, '*');
      }
    }

    // 纯原生 JS 实现点击切换，摆脱对主页面 jQuery 的高度依赖，避免跨域 DOM 操作报错
    document.addEventListener('DOMContentLoaded', () => {
      const toggleBtn = document.querySelector('.nb-toggle-btn');
      const content = document.querySelector('.nb-content');
      const tabs = document.querySelectorAll('.nb-tab-label');
      const pages = document.querySelectorAll('.nb-page');

      // 展开/收起逻辑
      toggleBtn.addEventListener('click', () => {
        toggleBtn.classList.toggle('active');
        content.classList.toggle('active');
        // 动画延迟后通知父 iframe 调整高度
        setTimeout(updateIframeHeight, 50);
      });

      // 标签页切换逻辑
      tabs.forEach(tab => {
        tab.addEventListener('click', (e) => {
          const idx = e.target.getAttribute('data-tab');

          tabs.forEach(t => t.classList.remove('active'));
          pages.forEach(p => p.classList.remove('active'));

          e.target.classList.add('active');
          pages[idx].classList.add('active');

          setTimeout(updateIframeHeight, 50);
        });
      });

      // 初始化时汇报一次高度
      setTimeout(updateIframeHeight, 100);
    });
  </script>
