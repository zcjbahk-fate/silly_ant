      removeItem: function(key) {

    function safeSetObjPath(obj, path, value) {
      const keys = path.split('.');
      let curr = obj;
      for (let i = 0; i < keys.length - 1; i++) {
        if (!curr[keys[i]] || typeof curr[keys[i]] !== 'object') curr[keys[i]] = {};
        curr = curr[keys[i]];
      }
      curr[keys[keys.length - 1]] = value;
    }

    async function fetchAvatarConfig() {
      try {
        const url = 'https://cdn.jsdelivr.net/gh/ll11292631-droid/i-am-daddy@main/%E6%BB%A8%E8%8E%B2.json?t=' + new Date().getTime();

    function getCharImages(name, data, deferSave = false) {
      let cachedImg = charImgCache[name];
      let avatarImg = cachedImg || '';
      let galleryImgs = [];
      let conditionKey = '';
      let canReroll = false;
      let isUpdated = false;

      if (!onlineAvatarConfig) return { avatar: avatarImg, gallery: cachedImg ? [cachedImg] : [], canReroll: false, conditionKey: '', cacheUpdated: false };

      if (onlineAvatarConfig["精确匹配"] && onlineAvatarConfig["精确匹配"][name]) {
        let imgs = onlineAvatarConfig["精确匹配"][name];
        galleryImgs = Array.isArray(imgs) ? imgs : [imgs];
        canReroll = galleryImgs.length > 1;
        conditionKey = name;

        if (!avatarImg || !galleryImgs.includes(avatarImg)) {
           avatarImg = galleryImgs[0];
           charImgCache[name] = avatarImg;
           isUpdated = true;
           if (!deferSave) SafeStorage.setItem('mvu_char_img_cache', JSON.stringify(charImgCache));
        }
        return { avatar: avatarImg, gallery: galleryImgs, canReroll: canReroll, conditionKey: conditionKey, type: 'exact', cacheUpdated: isUpdated };
      }

      let cupStr = (data.罩杯 || '').toLowerCase();
      let cupSize = '中';
      if (/[a]/.test(cupStr) || /小/.test(cupStr)) cupSize = '小';
      else if (/[b-c]/.test(cupStr)) cupSize = '中';
      else if (/[d-z]/.test(cupStr) || /大|巨/.test(cupStr)) cupSize = '大';

      let hairStr = data.发型 || '';
      let hairType = '低马尾';
      const hairPriorities = ['低双马尾', '盘发', '麻花辫', '低马尾', '卷发', '侧马尾', '双马尾', '长发', '短发', '马尾'];
      for (let h of hairPriorities) {
        if (hairStr.includes(h)) { hairType = h; break; }
      }

      conditionKey = `${cupSize}, ${hairType}`;
      let fallbackPool = onlineAvatarConfig[conditionKey] || onlineAvatarConfig?.["条件匹配"]?.[conditionKey] || onlineAvatarConfig?.["无匹配"];

      if (fallbackPool && Array.isArray(fallbackPool) && fallbackPool.length > 0) {
        galleryImgs = fallbackPool;
        canReroll = galleryImgs.length > 1;
        if (!avatarImg || !galleryImgs.includes(avatarImg)) {
          avatarImg = galleryImgs[Math.floor(Math.random() * galleryImgs.length)];
          charImgCache[name] = avatarImg;
          isUpdated = true;
          if (!deferSave) SafeStorage.setItem('mvu_char_img_cache', JSON.stringify(charImgCache));
        }
      } else if (typeof fallbackPool === 'string') {
        galleryImgs = [fallbackPool];
        canReroll = false;
        if (!avatarImg || avatarImg !== fallbackPool) {
          avatarImg = fallbackPool;
          charImgCache[name] = avatarImg;
          isUpdated = true;
          if (!deferSave) SafeStorage.setItem('mvu_char_img_cache', JSON.stringify(charImgCache));
        }
      } else {
        if (cachedImg) galleryImgs = [cachedImg];
      }

      return { avatar: avatarImg, gallery: galleryImgs, canReroll: canReroll, conditionKey: conditionKey, type: 'fuzzy', cacheUpdated: isUpdated };
    }

             fallbackPool = onlineAvatarConfig?.["精确匹配"]?.[charName];
          } else {
             fallbackPool = onlineAvatarConfig?.[conditionKey] || onlineAvatarConfig?.["条件匹配"]?.[conditionKey] || onlineAvatarConfig?.["无匹配"];
          }

          if (fallbackPool && Array.isArray(fallbackPool) && fallbackPool.length > 0) {
              const currentImg = charImgCache[charName] || '';
              let currentIdx = fallbackPool.indexOf(currentImg);
              let nextIdx = (currentIdx + 1) % fallbackPool.length;
              charImgCache[charName] = fallbackPool[nextIdx];
              SafeStorage.setItem('mvu_char_img_cache', JSON.stringify(charImgCache));
          }
          const cardIdx = $(this).closest('.char-card').attr('id').split('-').pop();
          renderCharacterPage();
          $(`#char-card-${cardIdx} .char-avatar`).click();
      });
