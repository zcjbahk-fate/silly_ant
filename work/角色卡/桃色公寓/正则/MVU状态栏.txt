<script type="module">
  window.mvuState = window.mvuState || { a: false, b: false };

  function populateCharacterData() {
    const allVars = getAllVariables();
    if (!allVars) return;

    $('#val-time').text(_.get(allVars, 'stat_data.时间', ''));
    $('#val-loc').text(_.get(allVars, 'stat_data.地点', ''));
    $('#val-char').text(_.get(allVars, 'stat_data.正在交互角色', ''));
    
    let money = _.get(allVars, 'stat_data.房东状态.资金总额', 0);
    $('#val-money').text(money + '￥');

    const tenants = ['楚云舒', '楚云微', '沈曼卿', '沈洛薇', '温知夏', '许嘉霓', '钟可可', '苏清禾', '盛凌烟'];
    tenants.forEach(name => {
      let rent = _.get(allVars, `stat_data.房客状态.${name}.已上交租金`, 0);
      $(`#val-${name}-rent`).text(rent + '￥');
      $(`#val-${name}-drain`).text(_.get(allVars, `stat_data.房客状态.${name}.榨精次数`, 0));
      $(`#val-${name}-bbc`).text(_.get(allVars, `stat_data.房客状态.${name}.与黑人性交次数`, 0));
      $(`#val-${name}-doing`).text(_.get(allVars, `stat_data.房客状态.${name}.现在在做`, ''));
    });

    const collections = _.get(allVars, 'stat_data.房东状态.地下室收藏品', []);
    if (Array.isArray(collections) && collections.length > 0) {
      const listHtml = collections.map((item, index) => `<div class="mvu-row">${index + 1}. ${item}</div>`).join('');
      $('#val-collections').html(listHtml);
    } else {
      $('#val-collections').text('暂无收藏');
    }
  }

  function bindEvents() {
    $('#header-a').off('click').on('click', function() {
      window.mvuState.a = !window.mvuState.a;
      $('#content-a').toggle(window.mvuState.a);
    });
    $('#header-b').off('click').on('click', function() {
      window.mvuState.b = !window.mvuState.b;
      $('#content-b').toggle(window.mvuState.b);
    });

    $('#content-a').toggle(window.mvuState.a);
    $('#content-b').toggle(window.mvuState.b);
  }

  async function init() {
    await waitGlobalInitialized('Mvu');
    bindEvents();
    populateCharacterData();
    eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, () => {
      populateCharacterData();
    });
  }

  $(errorCatched(init));
</script>
