  function getAttitudeText(score) {
    if (score <= 2) return "极度蔑视";
    if (score <= 4) return "轻蔑索取";
    if (score <= 6) return "客套伪装";
    if (score <= 8) return "温柔掌控";
    return "深度依恋";
  }

  function getExposureText(val) {
    if (val <= 0) return "完美隐蔽";
    if (val <= 30) return "微露痕迹";
    if (val <= 70) return "抓包把柄";
    return "彻底公开";
  }

  function updateDashboard(allVars) {
    const time = _.get(allVars, "stat_data.时间", "2025年6月1日 08:00");
    const loc = _.get(allVars, "stat_data.地点", "桃色公寓 四楼办公室");
    const char = _.get(allVars, "stat_data.正在交互角色", "") || "暂无";
    const money = Number(_.get(allVars, "stat_data.房东状态.资金总额", 20000));

    $("#val-time").text(time);
    $("#val-loc").text(loc);
    $("#val-char").text(char);

    const $money = $("#val-money");
    $money.text(money.toLocaleString() + " ￥");
    $money.removeClass("fund-safe fund-warn fund-danger");
    if (money >= 15000) $money.addClass("fund-safe");
    else if (money >= 5000) $money.addClass("fund-warn");
    else $money.addClass("fund-danger");

    return { activeChar: char };
  }

  function renderTenantCards(allVars, filterFloor = "all", activeChar = "") {
    const container = $("#tenants-container");
    container.empty();

    const filtered = filterFloor === "all" ? TENANTS : TENANTS.filter(t => t.floor === filterFloor);

    filtered.forEach(t => {
      const tenantData = _.get(allVars, `stat_data.房客状态.${t.name}`, {});
      const rent = tenantData.已上交租金 || 0;
      const drain = tenantData.榨精次数 || 0;
      const bbc = tenantData.与黑人性交次数 || 0;
      const attitude = Math.min(10, Math.max(1, tenantData.对user态度等级 !== undefined ? tenantData.对user态度等级 : 5));
      const exposure = Math.min(100, Math.max(0, tenantData.媚黑暴露程度 !== undefined ? tenantData.媚黑暴露程度 : 0));
      const doing = tenantData.现在在做 || "在房间中休憩……";

      const isInteracting = activeChar === t.name;

      let badge = t.badge;
      let badgeClass = t.badgeClass;
      if (t.name === "楚云舒") {
        if (tenantData.处女状态 === false) {
          badge = "已开苞 · 恶堕女S";
          badgeClass = "badge-wife";
        }
      } else if (t.name === "楚云微") {
        if (tenantData.处女状态 === false) {
          badge = "已开苞 · 小恶魔";
          badgeClass = "badge-wife";
        }
      }

      const cardHtml = `
        <div class="tenant-card ${isInteracting ? "active-interacting" : ""}">
          <div class="tenant-header">
            <div class="tenant-meta">
              <span class="room-tag">${t.room}</span>
              <span class="tenant-name-text">${t.name}</span>
              <span style="font-size:10.5px; color:var(--text-muted); margin-left:3px;">${t.tag}</span>
            </div>
            <span class="tenant-badge ${badgeClass}">${badge}</span>
          </div>

          <div class="stats-grid">
            <div class="stat-box">
              <span class="stat-k">已交租金</span>
              <span class="stat-v">${rent} ￥</span>
            </div>
            <div class="stat-box">
              <span class="stat-k">榨精次数</span>
              <span class="stat-v" style="color:#ff9ff3;">${drain} 次</span>
            </div>
            <div class="stat-box">
              <span class="stat-k">黑人交媾</span>
              <span class="stat-v" style="color:#f3a683;">${bbc} 次</span>
            </div>
          </div>

          <div class="meter-container">
            <div class="meter-row">
              <span class="meter-label">态度等级</span>
              <div class="meter-track">
                <div class="meter-fill meter-fill-attitude" style="width: ${attitude * 10}%;"></div>
              </div>
              <span class="meter-val-text">${attitude}/10 <span style="font-size:9.5px; color:var(--text-muted);">(${getAttitudeText(attitude)})</span></span>
            </div>
            <div class="meter-row">
              <span class="meter-label">媚黑暴露</span>
              <div class="meter-track">
                <div class="meter-fill meter-fill-exposure" style="width: ${exposure}%;"></div>
              </div>
              <span class="meter-val-text">${exposure}% <span style="font-size:9.5px; color:var(--text-muted);">(${getExposureText(exposure)})</span></span>
            </div>
          </div>

          <div class="doing-bubble">
            <strong>📍 监控动态:</strong> ${doing}
          </div>
        </div>
      `;
      container.append(cardHtml);
    });
  }

  function renderCollections(allVars) {
    const list = _.get(allVars, "stat_data.房东状态.地下室收藏品", []);
    const container = $("#val-collections");
    container.empty();

    if (!Array.isArray(list) || list.length === 0) {
      container.html(`<div class="empty-collection">🔒 地下室恋物收藏室暂无藏品，等待调教收缴中……</div>`);
      return;
    }

    const shelf = $(`<div class="collection-shelf"></div>`);
    list.forEach((item, idx) => {
      let icon = "👠";
      if (item.includes("内裤") || item.includes("贴身")) icon = "👙";
      else if (item.includes("丝袜") || item.includes("肉丝") || item.includes("黑丝")) icon = "🧦";
      else if (item.includes("借条") || item.includes("协议")) icon = "📜";

      shelf.append(`
        <div class="collection-item">
          <div class="collection-icon">${icon}</div>
          <div class="collection-text"><strong>#${idx + 1}</strong> ${item}</div>
        </div>
      `);
    });
    container.append(shelf);
  }

  function refreshAll() {
    const allVars = getAllVariables();
    if (!allVars) return;

    const { activeChar } = updateDashboard(allVars);
    renderTenantCards(allVars, window.mvuApartmentState.activeTab, activeChar);
    renderCollections(allVars);
  }

  function setupEvents() {
    $("#terminal-toggle").off("click").on("click", function() {
      window.mvuApartmentState.isExpanded = !window.mvuApartmentState.isExpanded;
      $("#mvu-body").toggle(window.mvuApartmentState.isExpanded);
      $("#toggle-icon").toggleClass("expanded", window.mvuApartmentState.isExpanded);
    });

    $(".nav-tab").off("click").on("click", function() {
      const tab = $(this).data("tab");
      window.mvuApartmentState.activeTab = tab;

      $(".nav-tab").removeClass("active");
      $(this).addClass("active");

      if (tab === "collection") {
        $("#panel-tenants").hide();
        $("#panel-collection").show();
      } else {
        $("#panel-collection").hide();
        $("#panel-tenants").show();
        const allVars = getAllVariables();
        const activeChar = _.get(allVars, "stat_data.正在交互角色", "");
        renderTenantCards(allVars, tab, activeChar);
      }
    });

    // 恢复状态
    $("#mvu-body").toggle(window.mvuApartmentState.isExpanded);
    $("#toggle-icon").toggleClass("expanded", window.mvuApartmentState.isExpanded);
    $(`.nav-tab[data-tab="${window.mvuApartmentState.activeTab}"]`).addClass("active");
    if (window.mvuApartmentState.activeTab === "collection") {
      $("#panel-tenants").hide();
      $("#panel-collection").show();
    } else {
      $("#panel-collection").hide();
      $("#panel-tenants").show();
    }
  }

  async function init() {
    await waitGlobalInitialized("Mvu");
    setupEvents();
    refreshAll();
    eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, () => {
      refreshAll();
    });
  }

  $(errorCatched(init));
</script>
