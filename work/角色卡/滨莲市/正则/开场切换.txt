<script>
(() => {
  const TARGET_WORLDBOOK_NAME = '滨莲市2.0';

  const ST = window.parent && typeof window.parent.getWorldbookNames === 'function' ? window.parent : window;
  const ST_getWorldbookNames = ST.getWorldbookNames;
  const ST_updateWorldbookWith = ST.updateWorldbookWith;
  const ST_substitudeMacros = ST.substitudeMacros;

  const PARENT_DOC = window.parent && window.parent.document ? window.parent.document : document;
  const $ = window.parent && window.parent.$ ? window.parent.$ : window.$;

  if (!$) {
    console.error('[开场白面板] 缺少 jQuery 环境，无法初始化');
    return;
  }

  function htmlEscape(text) {
    return String(text)
      .replace(/&/g, '\x26amp;')
      .replace(/</g, '\x26lt;')
      .replace(/>/g, '\x26gt;')
      .replace(/"/g, '\x26quot;')
      .replace(/'/g, '\x26#39;');
  }

  function toastSuccess(msg) {
    try {
      const t = window.parent && window.parent.toastr ? window.parent.toastr : window.toastr;
      if (t && typeof t.success === 'function') return t.success(msg);
    } catch (e) {}
  }
  function toastError(msg) {
    try {
      const t = window.parent && window.parent.toastr ? window.parent.toastr : window.toastr;
      if (t && typeof t.error === 'function') return t.error(msg);
    } catch (e) {}
  }
  function toastInfo(msg) {
    try {
      const t = window.parent && window.parent.toastr ? window.parent.toastr : window.toastr;
      if (t && typeof t.info === 'function') return t.info(msg);
    } catch (e) {}
  }

  const currentScript = document.currentScript;
  const $root = currentScript ? $(currentScript).prev('.th-opui-container') : $('.th-opui-container').last();

  function setStatus(state, text) {
    $root.find('[data-th-opui-status]').attr('data-state', state);
    $root.find('[data-th-opui-status-text]').text(text);
  }

  function getSelected(group) {
    return $root.find('[data-th-opui-choice="' + group + '"][data-selected="1"]').first().attr('data-value') || '';
  }

  function setSelected(group, value) {
    $root.find('[data-th-opui-choice="' + group + '"]').attr('data-selected', '0');
    $root.find('[data-th-opui-choice="' + group + '"][data-value="' + value + '"]').attr('data-selected', '1');
  }

  /* 全量同步所有条目开关状态 */
  async function syncAllTogglesAtomic() {
    if (!ST_updateWorldbookWith) return;

    const openingVal = getSelected('opening');
    const corpseVal = getSelected('corpse');
    const modifyVal = getSelected('modify');

    setStatus('sync', '正在同步...');

    try {
      await ST_updateWorldbookWith(TARGET_WORLDBOOK_NAME, wb => {
        let arr = Array.isArray(wb) ? wb : [];

        arr.forEach(e => {
          if (!e || !e.name) return;

          /* 开局条目：三选一 */
          if (e.name === '开局1') e.enabled = (openingVal === '1');
          if (e.name === '开局2') e.enabled = (openingVal === '2');
          if (e.name === '开局3') e.enabled = (openingVal === '3');

          /* 过往角色速览：开局1和开局2时激活 */
          if (e.name === '过往角色速览') e.enabled = (openingVal === '1' || openingVal === '2');

          /* 过往角色速览2：开局3时激活 */
          if (e.name === '过往角色速览2') e.enabled = (openingVal === '3');

          /* 尸体生成条目：二选一 */
          if (e.name === '尸体生成指导') e.enabled = (corpseVal === 'normal');
          if (e.name === '尸体生成指导(净)') e.enabled = (corpseVal === 'pure');

          /* 改造风格条目：二选一 */
          if (e.name === '改造剧情指导(推荐)') e.enabled = (modifyVal === 'normal');
          if (e.name === '改造剧情指导(血腥)') e.enabled = (modifyVal === 'gore');
        });

        return arr;
      }, { render: 'debounced' });

      setStatus('ok', '已实时同步');
    } catch (e) {
      console.error('同步开关失败', e);
      setStatus('err', '同步失败');
    }
  }

  /* 初始化时读取世界书状态 */
  async function initData() {
    const names = ST_getWorldbookNames ? ST_getWorldbookNames() : [];
    if (!names || !names.includes(TARGET_WORLDBOOK_NAME)) {
      setStatus('err', '未找到世界书');
      return;
    }

    let openingVal = '1';
    let corpseVal = 'normal';
    let modifyVal = 'normal';

    if (ST_updateWorldbookWith) {
      await ST_updateWorldbookWith(TARGET_WORLDBOOK_NAME, wb => {
        const arr = Array.isArray(wb) ? wb : [];

        arr.forEach(e => {
          if (!e || !e.name) return;

          /* 检测开局状态 */
          if (e.name === '开局2' && e.enabled) openingVal = '2';
          if (e.name === '开局3' && e.enabled) openingVal = '3';

          /* 检测尸体生成状态 */
          if (e.name === '尸体生成指导(净)' && e.enabled) corpseVal = 'pure';

          /* 检测改造风格状态 */
          if (e.name === '改造剧情指导(血腥)' && e.enabled) modifyVal = 'gore';
        });

        return arr;
      });
    }

    setSelected('opening', openingVal);
    setSelected('corpse', corpseVal);
    setSelected('modify', modifyVal);

    setStatus('ok', '准备就绪');
  }

  /* 选项卡点击事件 */
  $root.on('click', '[data-th-opui-choice]', function(ev) {
    const $btn = $(ev.currentTarget);
    const group = $btn.attr('data-th-opui-choice');
    const val = $btn.attr('data-value');
    setSelected(group, val);
    syncAllTogglesAtomic();
  });

  /* 开始按钮逻辑 */
  $root.on('click', '[data-th-opui-confirm]', async function(ev) {
    const rawText = String($root.find('[data-th-opui-input]').val() || '').trim();
    if (!rawText) {
      toastInfo('哥哥，请先输入主角设定哦~');
      return setStatus('err', '设定为空');
    }

    setStatus('sync', '正在写入...');
    try {
      await syncAllTogglesAtomic();

      await ST_updateWorldbookWith(TARGET_WORLDBOOK_NAME, wb => {
        let arr = Array.isArray(wb) ? wb : [];
        const userEntryName = ST_substitudeMacros ? ST_substitudeMacros('User倾向') : 'User倾向';
        const wrappedText = '<这是主角设定>\n' + rawText + '\n</这是主角设定>';
        const userIdx = arr.findIndex(e => e && e.name === userEntryName);

        const userEntry = {
          name: userEntryName,
          enabled: true,
          strategy: { type: 'constant', keys: [], keys_secondary: { logic: 'and_any', keys: [] }, scan_depth: 'same_as_global' },
          position: { type: 'after_character_definition', order: 0 },
          content: wrappedText,
          probability: 100,
          recursion: { prevent_incoming: true, prevent_outgoing: true, delay_until: null },
        };

        if (userIdx === -1) arr.push(userEntry);
        else arr[userIdx] = Object.assign({}, arr[userIdx], userEntry);

        return arr;
      }, { render: 'debounced' });

      const promptText = rawText + '\n\n以上是User的人设, 仅将User姓名转述为剧情中的玩家信息录入, 并开始后续剧情。';

      const $sendTextarea = $('#send_textarea', PARENT_DOC);
      if ($sendTextarea.length) {
        $sendTextarea.val(promptText).trigger('input');

        const $sendBtn = $('#send_but', PARENT_DOC);
        if ($sendBtn.length) {
            $sendBtn.trigger('click');
            toastSuccess('写入成功并已自动发送！');
            setStatus('ok', '已发送');
        } else {
            toastSuccess('写入成功！请手动点击发送。');
            setStatus('ok', '请发送');
        }
      } else {
        toastInfo('写入成功，请手动发送。');
        setStatus('ok', '需手动发');
      }
    } catch (e) {
      console.error('写入失败', e);
      toastError('写入失败了，请检查控制台。');
      setStatus('err', '写入失败');
    }
  });

  /* 启动初始化 */
  initData();
})();
</script>
</body>
</html>
