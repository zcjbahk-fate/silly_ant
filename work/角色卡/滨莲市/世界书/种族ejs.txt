    '僵尸': ['僵尸', '符纸封印僵尸'],
    '寄生体': ['寄生体'],
    '哭泣天使': ['哭泣天使'],
    '孕尸': ['孕尸'],

  for (var entryName in raceMap) {

      var keywords = raceMap[entryName];
      if (matchChatMessages(keywords)) {
        racesToLoad.add(entryName);
      }
    }
  }

  // 路径B：使用 matchChatMessages 扫描正文是否提到角色名，读取变量关联激活
  if (characterList && typeof characterList === 'object') {

      if (characterList.hasOwnProperty(charName)) {
        // 可靠地检测正文是否提到该角色
        if (matchChatMessages([charName])) {
          var charData = characterList[charName];
          if (charData && charData.种族 && typeof charData.种族 === 'string') {
            var race = charData.种族.trim();

            if (race !== '' && race !== '人类' && race !== '待初始化') {
              var matchedDict = false;
              for (var eName in raceMap) {
                if (raceMap.hasOwnProperty(eName)) {

                  for (var r = 0; r < raceKws.length; r++) {

                racesToLoad.add(race);
              }
            }

        }
      }
    }
